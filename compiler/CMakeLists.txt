cmake_minimum_required(VERSION 3.20)
project(MerlinBackend LANGUAGES C CXX)
find_package(MLIR REQUIRED CONFIG)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-fno-rtti)
include_directories(${MLIR_INCLUDE_DIRS} ${LLVM_INCLUDE_DIRS})
add_definitions(${LLVM_DEFINITIONS})
add_executable(merlin-opt src/main.cpp src/kernel.cpp src/device.cpp src/conv.cpp src/other.cpp src/inventory.cpp src/host.cpp src/pool.cpp)
target_link_libraries(merlin-opt PRIVATE MLIRIR MLIRParser MLIRLLVMDialect MLIRTargetLLVMIRImport MLIRLLVMIRToLLVMTranslation LLVMSupport LLVMIRReader LLVMCore LLVMTransformUtils LLVMAnalysis)
